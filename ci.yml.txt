name: CI

on:
  push:
  pull_request:

jobs:
  test:
    runs-on: ubuntu-latest
    container: swift:6.0
    steps:
      - uses: actions/checkout@v4
      - name: Build
        run: swift build --target RiftPatrolCore --target RiftPatrolSim
      - name: Test
        run: swift test
      - name: Balance simulation
        run: swift run RiftPatrolSim 20
